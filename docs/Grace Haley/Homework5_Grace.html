Question 3

Function that allows the user to query the News API by specifying a subject, a starting date, and an API key:

news_search <- function(subject, start_date, api_key) {
  
  response <- GET(
    "https://newsapi.org/v2/everything",
    query = list(
      q = subject,
      from = start_date,
      language = "en",
      pageSize = 50,
      apiKey = api_key
    )
  )
  
  news_content <- content(response, as = "text")
  news_content <- jsonlite::fromJSON(news_content)
  
  return(news_content$articles)
}
