  1. This function can be called with the 3 parameters to get information about a certain topic.
get_news <- function(topic, start_date, api_key) {
  
  response <- GET(
    "https://newsapi.org/v2/everything",
    query = list(
      q = topic,
      from = start_date,
      language = "en",
      pageSize = 50,
      apiKey = key
    )
  )
  
  news_content <- content(response)
  articles_df <- bind_rows(news_content$articles)
  return(articles_df)
}
