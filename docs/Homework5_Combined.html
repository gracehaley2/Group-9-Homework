get_news <- function(subject, from_date, api_key) {
  
  subject <- URLencode(subject)

    "https://newsapi.org/v2/everything?q=", subject,
    "&from=", from_date,
    "&language=en&pageSize=50&apiKey=", api_key
  )
  
  news <- GET(url)
  
  news_content <- content(news, simplifyDataFrame = TRUE)
  
  news_content$articles
}
